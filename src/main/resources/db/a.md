# Não faz nenhuma alteração nem verifica a estrutura do banco.
spring.jpa.hibernate.ddl-auto=none

# Apenas verifica se as tabelas do banco correspondem às entidades Java.
spring.jpa.hibernate.ddl-auto=validate

# Cria ou altera as tabelas automaticamente conforme as entidades Java.
spring.jpa.hibernate.ddl-auto=update

# Apaga e cria as tabelas sempre que a aplicação iniciar.
spring.jpa.hibernate.ddl-auto=create

# Cria as tabelas ao iniciar e apaga ao encerrar a aplicação.
spring.jpa.hibernate.ddl-auto=create-drop
